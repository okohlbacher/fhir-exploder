    import { useLocalStorage } from '@mantine/hooks';
    import { notifications } from '@mantine/notifications';
    import { useCallback, useEffect, useMemo, useState } from 'react';
    import {
      TRENDS_STORAGE_KEY,
      type QualitySnapshot,
    } from '../quality/trendsHistory';

    export interface UseTrendsHistoryReturn {
      snapshots: QualitySnapshot[];
      hydrated: boolean;
      append: (snap: QualitySnapshot) => void;
      clearAll: () => void;
    }

    export function useTrendsHistory(): UseTrendsHistoryReturn {
      const [stored, setStored] = useLocalStorage<QualitySnapshot[]>({
        key: TRENDS_STORAGE_KEY,
        defaultValue: [],
      });
      const [hydrated, setHydrated] = useState(false);
      useEffect(() => { setHydrated(true); }, []);

      // Defensive: localStorage could contain non-array (corrupt). Warn once, coerce.
      const warnedRef = /* use useRef(false) */ ...;
      const snapshots = useMemo(() => {
        if (Array.isArray(stored)) return stored;
        if (!warnedRef.current) {
          console.warn('quality.trends.v1 corrupted, resetting to []');
          warnedRef.current = true;
        }
        return [];
      }, [stored]);

      const append = useCallback((snap: QualitySnapshot) => {
        try {
          setStored((prev) => [...(Array.isArray(prev) ? prev : []), snap]);
        } catch (err) {
          if (err instanceof DOMException && err.name === 'QuotaExceededError') {
            notifications.show({
              color: 'red',
              title: 'Snapshot not saved',
              message: 'Browser storage is full. Clear history on the Trends tab to make room.',
              autoClose: 6000,
            });
          } else {
            throw err;
          }
        }
      }, [setStored]);

      const clearAll = useCallback(() => setStored([]), [setStored]);

      return { snapshots, hydrated, append, clearAll };
    }
    ```

      function BreachDot(props: { cx?: number; cy?: number; payload?: TrendPoint; includeOtherServers: boolean }) {
        const { cx, cy, payload, includeOtherServers } = props;
        if (cx == null || cy == null || payload == null || payload.score == null) return null;
        const fill = payload.threshold == null
          ? 'var(--mantine-color-gray-5)'
          : payload.breached
            ? 'var(--mantine-color-red-6)'
            : 'var(--mantine-color-blue-6)';
        // When only one server is relevant (filter off OR all snapshots share one server),
        // always render the default filled circle — no shape variation.
        const shapeIdx = includeOtherServers ? (payload.serverIndex ?? 0) % 4 : 0;
        const common = { stroke: 'white', strokeWidth: 1, 'data-server-shape': shapeIdx } as const;
        if (shapeIdx === 0) {
          return <circle cx={cx} cy={cy} r={4} fill={fill} {...common} />;
        }
        if (shapeIdx === 1) {
          // outlined square
          return <rect x={cx - 4} y={cy - 4} width={8} height={8} fill="white" stroke={fill} strokeWidth={1.5} data-server-shape={shapeIdx} />;
        }
        if (shapeIdx === 2) {
          // filled triangle (point up)
          return <polygon points={`${cx},${cy - 5} ${cx - 4.5},${cy + 3.5} ${cx + 4.5},${cy + 3.5}`} fill={fill} stroke="white" strokeWidth={1} data-server-shape={shapeIdx} />;
        }
        // outlined diamond (rhombus)
        return <polygon points={`${cx},${cy - 5} ${cx + 5},${cy} ${cx},${cy + 5} ${cx - 5},${cy}`} fill="white" stroke={fill} strokeWidth={1.5} data-server-shape={shapeIdx} />;
      }
      ```
