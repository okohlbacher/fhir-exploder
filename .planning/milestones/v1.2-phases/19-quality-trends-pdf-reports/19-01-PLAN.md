    ```typescript
    import type { MetricKey } from './thresholds';

    export const TRENDS_STORAGE_KEY = 'quality.trends.v1';  // per D-03
    export const TRENDS_SOFT_LIMIT = 500;                   // per D-04

    export interface QualitySnapshot {
      id: string;
      capturedAt: string;
      serverUrl: string;
      sampleSize: number;
      cohort: string[];
      scores: Record<MetricKey, number | null>;
      thresholds: Record<MetricKey, number | null>;
    }

    export interface MetricsReadSource {
      overallCompleteness: number | undefined;
      overallCoverage: number | undefined;
      overallValidation: number | undefined;
      overallPlausibility: number | undefined;
      overallLabRanges: number | undefined;
      overallDuplicates: number | undefined;
      overallReferences: number | undefined;
    }

    export interface CaptureSnapshotParams {
      metrics: MetricsReadSource;
      serverUrl: string;
      sampleSize: number;
      cohort: string[];
      getActiveThreshold: (key: MetricKey) => number | null;
    }

    const METRIC_KEYS: readonly MetricKey[] = [
      'completeness', 'coverage', 'validation', 'plausibility',
      'labRanges', 'duplicates', 'references',
    ] as const;

    export function captureSnapshot(params: CaptureSnapshotParams): QualitySnapshot {
      const { metrics, serverUrl, sampleSize, cohort, getActiveThreshold } = params;
      const rawScores: Record<MetricKey, number | undefined> = {
        completeness: metrics.overallCompleteness,
        coverage: metrics.overallCoverage,
        validation: metrics.overallValidation,
        plausibility: metrics.overallPlausibility,
        labRanges: metrics.overallLabRanges,
        duplicates: metrics.overallDuplicates,
        references: metrics.overallReferences,
      };
      const scores = Object.fromEntries(
        METRIC_KEYS.map((k) => [k, rawScores[k] ?? null]),
      ) as Record<MetricKey, number | null>;
      const thresholds = Object.fromEntries(
        METRIC_KEYS.map((k) => [k, getActiveThreshold(k)]),
      ) as Record<MetricKey, number | null>;
      return {
        id: crypto.randomUUID(),
        capturedAt: new Date().toISOString(),
        serverUrl,
        sampleSize,
        cohort: [...cohort],  // defensive copy — caller's array must not alias storage
        scores,
        thresholds,
      };
    }

    export function filterSnapshotsByServer(
      snapshots: QualitySnapshot[],
      serverUrl: string,
    ): QualitySnapshot[] {
      return snapshots.filter((s) => s.serverUrl === serverUrl);
    }

    export function serverUrlSlug(url: string): string {
      const cleaned = url
        .replace(/^https?:\/\//i, '')
        .replace(/\/fhir\/?$/i, '')
        .replace(/[:/\\?#&]+/g, '-')
        .replace(/^-+|-+$/g, '')
        .slice(0, 60);
      return cleaned || 'unknown-server';
    }

    export function pdfFilename(serverUrl: string, at: Date): string {
      const iso = at.toISOString();                             // 2026-04-14T18:30:42.123Z
      const datePart = iso.slice(0, 10);                        // 2026-04-14
      const timePart = iso.slice(11, 19).replace(/:/g, '');     // 183042
      return `fhir-exploder-quality-report_${serverUrlSlug(serverUrl)}_${datePart}_${timePart}.pdf`;
    }

    export function computeBreachedFlag(
      score: number | null | undefined,
      threshold: number | null,
    ): boolean {
      if (score == null) return false;
      if (threshold == null) return false;
      return score < threshold;
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser `localStorage` ↔ app | `quality.trends.v1` JSON read on load. Corrupt payload (user / malicious extension edit) must not crash the app. |
| User-supplied `serverUrl` (from settings.yaml or Blaze config) ↔ filesystem (via PDF download filename) | The `serverUrl` flows into the download filename; path separators / shell chars must not survive. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-19-01 | Tampering | `quality.trends.v1` localStorage read | mitigate | Plan 02's `useTrendsHistory` wraps reads in `Array.isArray`; corrupt payload → `[]` + `console.warn`. This plan's `trendsHistory.ts` contains no direct localStorage read, so mitigation is deferred to Plan 02. Noted here for traceability — Plan 01 is type/helper layer only. |
| T-19-02 | Tampering (filesystem injection) | `serverUrlSlug` → download filename | mitigate | `serverUrlSlug` regex strips `[:/\\?#&]+` and trims to 60 chars; empty string → `'unknown-server'`. Test `pdf-filename.test.ts` asserts no forbidden char survives across 5+ adversarial URL inputs. |
| T-19-03 | Information Disclosure | `QualitySnapshot.serverUrl` stored in localStorage | accept | Local-first tool, no multi-user context; serverUrl is already visible in the settings UI. Same exposure surface as existing `quality.cohort.v1` / `quality.thresholds.v1`. |
| T-19-04 | Information Disclosure | `QualitySnapshot.scores` + `thresholds` (percentages only, no PHI) | accept | Scores are aggregate percentages (0–100), no patient-level data. No PHI in the snapshot payload by design (D-02 locks the payload shape — deliberately excludes resource IDs / patient fields). |
| T-19-05 | DoS (self-inflicted) | Unbounded snapshot array in localStorage (`QuotaExceededError`) | mitigate | `TRENDS_SOFT_LIMIT = 500` exported here; soft warning + toast surfacing is Plan 02's responsibility. |
| T-19-06 | Repudiation | Snapshot timestamp tampering | accept | User's own browser; no audit requirement. Timestamp is `new Date().toISOString()` at capture, good-faith only. |
</threat_model>
