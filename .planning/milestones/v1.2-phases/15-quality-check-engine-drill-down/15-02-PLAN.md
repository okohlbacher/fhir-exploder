2. **Constants:**
```typescript
const PAGE_SIZE = 50;

const SEVERITY_COLOR: Record<IssueSeverity, string> = {
  error: 'red',
  warning: 'yellow',
  info: 'blue',
};

const SEVERITY_ORDER: Record<IssueSeverity, number> = {
  error: 0,
  warning: 1,
  info: 2,
};

const SEVERITY_OPTIONS = ['All severities', 'error', 'warning', 'info'];
```

3. **Props interface:**
```typescript
export interface ResourceIssueTableProps {
  issues: NormalizedIssue[];
  initialFieldFilter?: string;
}
```

4. **Component body:**
- State: `page` (number, default 1), `severityFilter` (string, default 'All severities'), `fieldFilter` (string, default from `initialFieldFilter ?? ''`)
- When `initialFieldFilter` changes, update fieldFilter state: `useEffect(() => { if (initialFieldFilter !== undefined) setFieldFilter(initialFieldFilter); }, [initialFieldFilter])`
- `filtered` memo: filter issues by severity (skip if 'All severities') and field path (case-insensitive includes), then sort by SEVERITY_ORDER then resourceId.localeCompare
- `totalPages = Math.ceil(filtered.length / PAGE_SIZE)`
- `pageItems = filtered.slice((page - 1) * PAGE_SIZE, page * PAGE_SIZE)`
- Reset page to 1 when severityFilter or fieldFilter changes (in the onChange handlers)
- Start/end display numbers: `start = (page - 1) * PAGE_SIZE + 1`, `end = Math.min(page * PAGE_SIZE, filtered.length)`

5. **Empty state (no issues at all):**
```tsx
if (issues.length === 0) {
```

6. **Filter bar:**
```tsx
<Group gap="sm" mb="sm">
  <Select
    data={SEVERITY_OPTIONS}
    value={severityFilter}
    onChange={(v) => { setSeverityFilter(v ?? 'All severities'); setPage(1); }}
    size="sm"
    w={180}
    aria-label="Filter by severity"
  />
```

9. **Table:** Matches ValidationIssueList structure:
```tsx
<div style={{ maxHeight: 600, overflow: 'auto' }}>
  <Table striped highlightOnHover stickyHeader>
    <Table.Thead>
      <Table.Tr>
        <Table.Th>#</Table.Th>
        <Table.Th>Severity</Table.Th>
        <Table.Th>Resource</Table.Th>
        <Table.Th>Field</Table.Th>
        <Table.Th>Description</Table.Th>
      </Table.Tr>
    </Table.Thead>
    <Table.Tbody>
      {pageItems.map((issue, i) => {
        const parts = issue.resourceId.split('/');
        const type = parts[0];
        const id = parts.slice(1).join('/');
        const rowNum = (page - 1) * PAGE_SIZE + i + 1;
        return (
          <Table.Tr key={`${issue.resourceId}|${issue.field}|${i}`}>
            <Table.Td>{rowNum}</Table.Td>
            <Table.Td>
              <Badge color={SEVERITY_COLOR[issue.severity]} variant="light" size="sm">
                {issue.severity}
              </Badge>
            </Table.Td>
            <Table.Td>
              {type && id ? (
                <Anchor component={Link} to={`/explorer/${type}/${id}`} c="blue.6" size="sm">
                  {issue.resourceId}
                </Anchor>
              ) : (
                <Code>--</Code>
              )}
            </Table.Td>
            <Table.Td><Code>{issue.field}</Code></Table.Td>
            <Table.Td>{issue.description}</Table.Td>
          </Table.Tr>
        );
      })}
    </Table.Tbody>
  </Table>
</div>
```
