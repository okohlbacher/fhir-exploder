    ```ts
    /**
     * Sampling helper used by quality panels...
     *
     * Phase 21 (21-03): optional `patientIds` parameter scopes results to a
     * patient set. Uses GET ?patient=... for ≤40 IDs, POST _search for larger
     * cohorts (Pitfall 2 — URL length ceiling). `Patient` type uses `_id=`
     * since `patient=` isn't valid on Patient (Pitfall 4).
     */
    import type { MedplumClient } from '@medplum/core';
    import type { Bundle, Resource, ResourceType } from '@medplum/fhirtypes';

    export const SHORT_QUERY_THRESHOLD = 40;

    export async function sampleResources(
      client: MedplumClient,
      resourceType: string,
      sampleSize: number,
      patientIds?: string[],
    ): Promise<Resource[]> {
      const hasCohort = patientIds && patientIds.length > 0;
      if (!hasCohort) {
        return client.searchResources(resourceType as ResourceType, { _count: String(sampleSize) });
      }
      const isPatientType = resourceType === 'Patient';
      const scopeParam = isPatientType ? '_id' : 'patient';
      const scopeValue = isPatientType
        ? patientIds!.join(',')
        : patientIds!.map((id) => `Patient/${id}`).join(',');

      if (patientIds!.length <= SHORT_QUERY_THRESHOLD) {
        return client.searchResources(resourceType as ResourceType, {
          _count: String(sampleSize),
          [scopeParam]: scopeValue,
        });
      }
      const body = new URLSearchParams({
        _count: String(sampleSize),
        [scopeParam]: scopeValue,
      }).toString();
      const bundle = (await client.post(
        client.fhirUrl(resourceType, '_search').toString(),
        body,
        'application/x-www-form-urlencoded',
      )) as Bundle<Resource>;
      return (bundle.entry ?? []).map((e) => e.resource!).filter(Boolean);
    }
    ```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-21-07 | Injection (FHIR URL/body) | `sampleResources` URL/body construction | mitigate | GET path: params passed as object → Medplum serializes via URLSearchParams internally. POST path: `new URLSearchParams({...}).toString()` URL-encodes all values. No string concatenation into the URL path. Threshold constant 40 prevents URL-length ambiguity. |
| T-21-08 | DoS (URL length overflow) | GET query for large cohorts | mitigate | Hard threshold at 40 IDs (~1.9KB URL under Jetty's 8KB default); >40 → POST _search (spec-sanctioned escape hatch). Test `long POST` enforces. |
