```typescript
// src/utils/searchByIdentifierPrefix.ts
import { MedplumClient } from '@medplum/core';
import { Bundle, ResourceType } from '@medplum/fhirtypes';

export interface SearchByIdentifierPrefixOptions {
  /** Max IDs to enumerate via _elements=id&_count= (was MAX_ID_FETCH). Default 5000. */
  limit?: number;
  /** Paginated result batch size for the follow-up _id=a,b,c fetch. Default 20. */
  pageSize?: number;
}

export async function searchByIdentifierPrefix<T extends ResourceType>(
  client: MedplumClient,
  type: T,
  prefix: string,
  options?: SearchByIdentifierPrefixOptions,
): Promise<Bundle>;
```

Semantics (lifted verbatim from RESEARCH.md §Focus 2):
1. GET `${type}?_elements=id&_count=${limit}` — enumerate IDs
2. Filter `bundle.entry[].resource.id` by `startsWith(prefix)` -> `matching`
3. If `matching.length === 0`: return `{ resourceType: 'Bundle', type: 'searchset', total: 0, entry: [] }`
4. Else: `pageIds = matching.slice(0, pageSize)`; GET `${type}?_id=${pageIds.join(',')}&_count=${pageSize}` -> `result`
5. Set `result.total = matching.length` (preserves "Override total to reflect all matches, not just this page" — SearchResultsPage.tsx:194 comment is load-bearing)
6. Return `result`
