From 24-RESEARCH.md §Pattern 2 — `asyncRunReducer.ts`:
```typescript
export type AsyncRunStatus = 'idle' | 'running' | 'complete' | 'cancelled' | 'error';

export interface AsyncRunState<TIssue> {
  status: AsyncRunStatus;
  progress: { current: number; total: number };
  issues: TIssue[];
  errorMessage?: string;
}

export type AsyncRunAction<TIssue> =
  | { type: 'start' }
  | { type: 'progress'; current: number; total: number }
  | { type: 'append-issues'; issues: TIssue[] }
  | { type: 'complete' }
  | { type: 'cancel' }
  | { type: 'error'; message: string }
  | { type: 'reset' };

export function asyncRunReducer<TIssue>(
  state: AsyncRunState<TIssue>,
  action: AsyncRunAction<TIssue>,
): AsyncRunState<TIssue>;

export function initialAsyncRunState<TIssue>(): AsyncRunState<TIssue>;
```

Reducer transition table (authoritative):
| action | precondition | next state |
|--------|--------------|------------|
| `start` | any | `{status: 'running', progress: {0,0}, issues: [], errorMessage: undefined}` |
| `progress` | any | `{...state, progress: {current: action.current, total: action.total}}` |
| `append-issues` | any | `{...state, issues: [...state.issues, ...action.issues]}` |
| `complete` | `state.status === 'cancelled'` | `state` (no-op — cancel wins) |
| `complete` | else | `{...state, status: 'complete'}` |
| `cancel` | `state.status === 'running'` | `{...state, status: 'cancelled'}` |
| `cancel` | else | `state` (idempotent) |
| `error` | any | `{...state, status: 'error', errorMessage: action.message}` |
| `reset` | any | `{status: 'idle', progress: {0,0}, issues: [], errorMessage: undefined}` |

From 24-RESEARCH.md §Pattern 2 — `useAsyncRun.ts` exported API:
```typescript
interface RunnerHelpers<TIssue> {
  isCancelled: () => boolean;
  setProgress: (current: number, total: number) => void;
  appendIssues: (batch: TIssue[]) => void;
}

interface UseAsyncRunArgs<TIssue> {
  runner: (helpers: RunnerHelpers<TIssue>) => Promise<void>;
  deps?: unknown[];
  autoStart?: boolean;
}

export interface UseAsyncRunResult<TIssue> extends AsyncRunState<TIssue> {
  start: () => void;
  cancel: () => void;
}

export function useAsyncRun<TIssue>(args: UseAsyncRunArgs<TIssue>): UseAsyncRunResult<TIssue>;
```

Cancellation pattern (load-bearing):
- Each call to `start()` creates a FRESH `let cancelled = false` in that call's closure.
- A `cancelInFlightRef = useRef<(() => void) | null>(null)` holds a function that flips the CURRENT run's `cancelled` to `true`. On each `start()`, the previous ref's stored function is invoked first to cancel the prior run, then replaced with a new closure that flips THIS run's flag.
- This is NOT the `cancelledRef` anti-pattern — the ref holds a function, not a boolean. Document this rationale in the file header, referencing `useCompletenessReport.ts:71-74`.
