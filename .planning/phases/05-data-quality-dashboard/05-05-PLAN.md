```typescript
import type { Resource, StructureDefinition, OperationOutcomeIssue } from '@medplum/fhirtypes';
import type { ValidationBackend } from './types';
import { requiredElementPaths, isPathPopulated } from './completenessWalker';

export function validateStructural(
  resource: Resource,
  profile: StructureDefinition | null,
): OperationOutcomeIssue[] {
  if (!profile) return [];
  const required = requiredElementPaths(profile);
  const issues: OperationOutcomeIssue[] = [];
  for (const path of required) {
    if (!isPathPopulated(resource, path)) {
      issues.push({
        severity: 'error',
        code: 'required',
        expression: [path],
        diagnostics: `${path} is required by ${profile.name ?? profile.url} but was not populated`,
      });
    }
  }
  return issues;
}

export function createStructuralBackend(
  getProfile: (resourceType: string) => StructureDefinition | null,
): ValidationBackend {
  return {
    kind: 'structural',
    async validate(resource) {
      return validateStructural(resource, getProfile(resource.resourceType));
    },
  };
}
```

```typescript
import { MedplumClient } from '@medplum/core';
import type { OperationOutcome, OperationOutcomeIssue, Resource } from '@medplum/fhirtypes';
import type { ValidationBackend } from './types';

export function createValidatorClient(validatorUrl: string): MedplumClient {
  // Mirror src/terminology/terminologyClient.ts pattern — second client pointed at non-Blaze URL
  return new MedplumClient({
    baseUrl: validatorUrl,
    fhirUrlPath: '',  // validatorUrl is already the FHIR root
    fetch: (input, init) => fetch(input, init),
  });
}

export function createRemoteBackend(
  validatorUrl: string,
  profileCanonical?: string,
): ValidationBackend {
  const client = createValidatorClient(validatorUrl);
  return {
    kind: 'remote',
    async validate(resource: Resource): Promise<OperationOutcomeIssue[]> {
      const path = `${resource.resourceType}/$validate${profileCanonical ? `?profile=${encodeURIComponent(profileCanonical)}` : ''}`;
      try {
        const outcome = await client.post<OperationOutcome>(path, resource);
        return outcome?.issue ?? [];
      } catch (err) {
        // Surface as a single information-severity issue so the UI can display it inline
        return [{
          severity: 'error',
          code: 'exception',
          diagnostics: `Remote validator (${validatorUrl}) failed: ${(err as Error).message}`,
        }];
      }
    },
  };
}
```

```typescript
import type { AppSettings } from '../config/types';
import type { OperationOutcomeIssue } from '@medplum/fhirtypes';
import type { ValidationBackend } from './types';
import { createStructuralBackend } from './structuralValidator';
import { createRemoteBackend } from './remoteValidator';
import { getProfileForType } from './profiles';

export interface BackendResolution {
  backends: ValidationBackend[];
  hasRemote: boolean;
  hasProfile: boolean;
  validatorUrl: string | null;
}

export function resolveBackends(settings: AppSettings, resourceType: string): BackendResolution {
  const profile = getProfileForType(resourceType);
  const hasProfile = profile !== null;
  const validatorUrl = settings.validation?.validatorUrl ?? null;
  const hasRemote = typeof validatorUrl === 'string' && validatorUrl.length > 0;
  const backends: ValidationBackend[] = [createStructuralBackend(getProfileForType)];
  if (hasRemote) {
    backends.push(createRemoteBackend(validatorUrl!, profile?.url));
  }
  return { backends, hasRemote, hasProfile, validatorUrl };
}

export function dedupeIssues(issues: OperationOutcomeIssue[]): OperationOutcomeIssue[] {
  const seen = new Set<string>();
  const out: OperationOutcomeIssue[] = [];
  for (const issue of issues) {
    const key = `${issue.severity}|${issue.code}|${JSON.stringify(issue.expression ?? issue.location ?? [])}|${issue.diagnostics ?? ''}`;
    if (seen.has(key)) continue;
    seen.add(key);
    out.push(issue);
  }
  return out;
}
```

**1. `src/hooks/useValidationRun.ts`** — batch runner state machine:

```typescript
import { useState, useRef, useCallback } from 'react';
import type { MedplumClient } from '@medplum/core';
import type { OperationOutcomeIssue } from '@medplum/fhirtypes';
import type { AppSettings } from '../config/types';
import { sampleResources } from '../quality/sampling';
import { resolveBackends, dedupeIssues } from '../quality/validationBackends';

export type ValidationRunStatus = 'idle' | 'running' | 'complete' | 'cancelled' | 'error';

export interface ValidationRunState {
  status: ValidationRunStatus;
  progress: { current: number; total: number };
  issues: OperationOutcomeIssue[];   // flattened, deduped across resources
  byResource: Record<string, { resourceId: string; issues: OperationOutcomeIssue[] }>;
  errorMessage?: string;
  start: () => void;
  cancel: () => void;
}

interface UseValidationRunArgs {
  client: MedplumClient;
  resourceType: string;
  sampleSize: number;
  batchSize: number;
  settings: AppSettings;
}

export function useValidationRun({
  client, resourceType, sampleSize, batchSize, settings,
}: UseValidationRunArgs): ValidationRunState {
  const [status, setStatus] = useState<ValidationRunStatus>('idle');
  const [progress, setProgress] = useState({ current: 0, total: 0 });
  const [issues, setIssues] = useState<OperationOutcomeIssue[]>([]);
  const [byResource, setByResource] = useState<Record<string, { resourceId: string; issues: OperationOutcomeIssue[] }>>({});
  const [errorMessage, setErrorMessage] = useState<string | undefined>();
  const cancelledRef = useRef(false);

  const start = useCallback(async () => {
    cancelledRef.current = false;
    setStatus('running');
    setIssues([]);
    setByResource({});
    setErrorMessage(undefined);
    try {
      const sample = await sampleResources(client, resourceType, sampleSize);
      setProgress({ current: 0, total: sample.length });
      const { backends } = resolveBackends(settings, resourceType);

      for (let i = 0; i < sample.length; i += batchSize) {
        if (cancelledRef.current) { setStatus('cancelled'); return; }
        const batch = sample.slice(i, i + batchSize);
        const results = await Promise.all(batch.map(async (r) => {
          const perBackend = await Promise.all(backends.map(b => b.validate(r)));
          const deduped = dedupeIssues(perBackend.flat());
          return { resource: r, issues: deduped };
        }));
        if (cancelledRef.current) { setStatus('cancelled'); return; }
        setIssues(prev => [...prev, ...results.flatMap(r => r.issues.map(i => ({ ...i, _resourceId: `${r.resource.resourceType}/${r.resource.id}` } as OperationOutcomeIssue)))]);
        setByResource(prev => {
          const next = { ...prev };
          for (const r of results) {
            const key = `${r.resource.resourceType}/${r.resource.id}`;
            next[key] = { resourceId: r.resource.id ?? 'unknown', issues: r.issues };
          }
          return next;
        });
        setProgress({ current: Math.min(i + batch.length, sample.length), total: sample.length });
      }
      if (!cancelledRef.current) setStatus('complete');
    } catch (err) {
      setStatus('error');
      setErrorMessage((err as Error).message);
    }
  }, [client, resourceType, sampleSize, batchSize, settings]);

  const cancel = useCallback(() => { cancelledRef.current = true; }, []);

  return { status, progress, issues, byResource, errorMessage, start, cancel };
}
```

```typescript
import { Badge, Code, Table, Anchor } from '@mantine/core';
import { Link } from 'react-router-dom';
import type { OperationOutcomeIssue } from '@medplum/fhirtypes';

const SEVERITY_COLOR: Record<string, string> = {
  fatal: 'red', error: 'red', warning: 'yellow', information: 'blue',
};
const SEVERITY_ORDER: Record<string, number> = {
  fatal: 0, error: 1, warning: 2, information: 3,
};

export interface ValidationIssueListProps {
  issues: Array<OperationOutcomeIssue & { _resourceId?: string }>;
}

export function ValidationIssueList({ issues }: ValidationIssueListProps) {
  const sorted = [...issues].sort((a, b) => {
    const s = (SEVERITY_ORDER[a.severity] ?? 99) - (SEVERITY_ORDER[b.severity] ?? 99);
    if (s !== 0) return s;
    return (a._resourceId ?? '').localeCompare(b._resourceId ?? '');
  });
  return (
    <div style={{ maxHeight: 600, overflow: 'auto' }}>
      <Table striped highlightOnHover stickyHeader>
        <Table.Thead>
          <Table.Tr>
            <Table.Th>#</Table.Th>
            <Table.Th>Severity</Table.Th>
            <Table.Th>Resource</Table.Th>
            <Table.Th>Location</Table.Th>
            <Table.Th>Issue</Table.Th>
          </Table.Tr>
        </Table.Thead>
        <Table.Tbody>
          {sorted.map((issue, i) => {
            const resourceId = issue._resourceId;

                    </Anchor>
                  ) : <Code>—</Code>}
                </Table.Td>
                <Table.Td>
```
