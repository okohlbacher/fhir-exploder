    export interface FhirpathCriterionCardProps {
      /** Initial expression (for Edit mode pre-fill). Empty string = no criterion. */
      initialExpression?: string;
      /**
       * Called when the user's expression changes. Parent should store the
       * current value as the source of truth for building the criteria array.
       * When the value is empty, the parent should OMIT the FHIRPath criterion

       */
      onExpressionChange: (expression: string) => void;
      /**
       * Called on successful validate with the translated FHIR search URL.
       * Parent should cache this on the FhirpathCriterion.translatedQuery
       * so the cohortResolver can consume it (though the resolver also
       * re-translates on every resolve for determinism — see Plan 22-01 Task 3).
       * Called with `undefined` when the user edits after a successful validate
       * (cache invalidation signal to parent).
       */
      onValidated: (translatedQuery: string | undefined) => void;
    }

    Implementation skeleton follows UI-SPEC §S1 JSX sketch lines 429-484 VERBATIM. Key behaviours:

    ```typescript
    import {
      Paper, Stack, Title, Text, Textarea, Button, Group, Anchor, Collapse,
      Code, Loader,
    } from '@mantine/core';
    import {
      IconCheck, IconAlertTriangle,
    } from '@tabler/icons-react';
    import { useMedplum } from '@medplum/react';
    import { useState, useRef, useCallback } from 'react';
    import {
      translateFhirpath,
      translatedQueryToFhirSearchUrl,
      dryRunCount,
      TranslationError,
    } from '../../quality/fhirpathTranslator';

    type ResultState =
      | { kind: 'pristine' }
      | { kind: 'validating' }
      | { kind: 'success'; count: number }
      | { kind: 'empty' }
      | { kind: 'truncated'; count: number }
      | { kind: 'translation-error'; message: string }
      | { kind: 'server-error' }
      | { kind: 'timeout' };

    export function FhirpathCriterionCard(props: FhirpathCriterionCardProps): JSX.Element {
      const medplum = useMedplum();
      const [expression, setExpression] = useState(props.initialExpression ?? '');
      const [result, setResult] = useState<ResultState>({ kind: 'pristine' });
      const [helperOpen, setHelperOpen] = useState(false);
      const lastValidatedRef = useRef<string>('');  // the expression that was last validated
      const abortRef = useRef<AbortController | null>(null);

      const handleExpressionChange = useCallback(
        (e: React.ChangeEvent<HTMLTextAreaElement>) => {
          const v = e.currentTarget.value;
          setExpression(v);
          props.onExpressionChange(v);
          // If user edited after a successful validate, invalidate cache + revert result row
          if (v !== lastValidatedRef.current) {
            if (result.kind === 'success' || result.kind === 'empty' || result.kind === 'truncated') {
              setResult({ kind: 'pristine' });
              props.onValidated(undefined);
            }
          }
        },
        [props, result.kind],
      );

      const handleValidate = useCallback(async () => {
        abortRef.current?.abort();
        const controller = new AbortController();
        abortRef.current = controller;
        setResult({ kind: 'validating' });
        const timeoutId = window.setTimeout(() => controller.abort(), 10_000);
        try {
          const tq = translateFhirpath(expression);
          const url = translatedQueryToFhirSearchUrl(tq);
          const n = await dryRunCount(medplum, tq);
          if (controller.signal.aborted) return;
          lastValidatedRef.current = expression;
          props.onValidated(url);
          if (n === 0) {
            setResult({ kind: 'empty' });
          } else if (n > 10_000) {
            setResult({ kind: 'truncated', count: n });
          } else {
            setResult({ kind: 'success', count: n });
          }
        } catch (err) {
          if (controller.signal.aborted) {
            setResult({ kind: 'timeout' });
            props.onValidated(undefined);
            return;
          }
          if (err instanceof TranslationError) {
            setResult({ kind: 'translation-error', message: err.message });
          } else {
            setResult({ kind: 'server-error' });
          }
          props.onValidated(undefined);
        } finally {
          window.clearTimeout(timeoutId);
          abortRef.current = null;
        }
      }, [expression, medplum, props]);

      const validating = result.kind === 'validating';

      return (
        <Paper withBorder radius="sm" p="md">
          <Stack gap="md">
            <Title order={4}>FHIRPath query (advanced)</Title>
            <Text size="sm" c="dimmed">
              Optional. Define this cohort's criterion as a FHIRPath expression. Combines with the criteria above using AND-intersection.
            </Text>
            <Textarea
              label="FHIRPath expression"
              placeholder="Patient.where(birthDate < @1960-01-01)"
              description={
                <>
                  Single <Code>Resource.where(field op literal)</Code> expression. Composition (and / or), nested where, and function calls (exists, first, count) are not supported. Maximum 4,096 characters.
                </>
              }
              autosize
              minRows={4}
              maxRows={12}
              maxLength={4096}
              style={{ fontFamily: 'monospace' }}
              value={expression}
              onChange={handleExpressionChange}
            />
            <Group justify="space-between" align="center" wrap="wrap">
              <Anchor
                component="button"
                type="button"
                onClick={() => setHelperOpen((o) => !o)}
                aria-expanded={helperOpen}
                aria-controls="fhirpath-helper-collapse"
              >
                {helperOpen ? 'Hide supported syntax and examples' : 'Show supported syntax and examples'}
              </Anchor>
              <Button
                variant="filled"
                color="blue"
                leftSection={<IconCheck size={16} />}
                loading={validating}
                disabled={expression.trim().length === 0 || validating}
                title={expression.trim().length === 0 ? 'Enter a FHIRPath expression to validate.' : undefined}
                onClick={handleValidate}
              >
                Validate
              </Button>
            </Group>
            <div role="status" aria-live="polite">
              {renderResultRow(result)}
            </div>
            <Collapse in={helperOpen} id="fhirpath-helper-collapse">
              {/* Helper content per UI-SPEC §S1 Copywriting table — 4 resource sub-headings with 2-3 Code examples each + footer note */}
            </Collapse>
          </Stack>
        </Paper>
      );
    }

    function renderResultRow(result: ResultState): React.ReactNode {
      switch (result.kind) {
        case 'pristine':
          return <Text size="xs" c="dimmed">Click Validate to check this expression.</Text>;
        case 'validating':
          return (
            <Group gap="xs">
              <Loader size="xs" />
              <Text size="xs" c="dimmed">Validating…</Text>
            </Group>
          );
        case 'success':
          return (
            <Group gap="xs">
              <IconCheck size={14} color="green" />
              <Text size="xs" c="green.7">Matches {result.count.toLocaleString()} patients.</Text>
            </Group>
          );
        case 'empty':
          return (
            <Group gap="xs">
              <IconAlertTriangle size={14} color="yellow" />
              <Text size="xs" c="yellow.8">Cohort would be empty (0 patients matched).</Text>
            </Group>
          );
        case 'truncated':
          return (
            <Group gap="xs">
              <IconAlertTriangle size={14} color="yellow" />
              <Text size="xs" c="yellow.8">Matches more than 10,000 patients — only the first 10,000 will be analyzed. Add more criteria to narrow the cohort.</Text>
            </Group>
          );
        case 'translation-error':
          return (
            <Group gap="xs">
              <IconAlertTriangle size={14} color="red" />
              <Text size="xs" c="red.7">{result.message}</Text>
            </Group>
          );
        case 'server-error':
          return <Text size="xs" c="red.7">Could not validate against the server. Check your connection or try again.</Text>;
        case 'timeout':
          return <Text size="xs" c="yellow.8">Validation timed out. Your expression may be valid but the server can't count results quickly. Try Save anyway, or refine the criterion.</Text>;
      }
    }
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| File picker → Import handler | Untrusted user-selected `File` object crossing into the browser's `file.text()` → codec pipeline |
| Modal input → `updateCohort` mutation | User-edited `name` + `criteria` payload flowing from form state into localStorage persistence |
| Cohort name → `downloadString` filename | User-controlled string used as a filename — must be slugified before disk write |

| Threat ID | Category | Component | Severity | Disposition | Mitigation Plan |
|-----------|----------|-----------|----------|-------------|-----------------|
| T-22-12 | Tampering / Injection | Export filename built from `cohort.name` — an attacker who can set a cohort name (e.g. via Import from a malicious FDPG file) could attempt path traversal or filename injection (`../../../secrets.json`) | LOW | mitigate | `slugifyCohortName` replaces every non-`[a-z0-9]+` run with `-`, trims leading/trailing dashes, and falls back to `'cohort'` on empty. Defence-in-depth: the browser's `download` attribute on `<a>` (used by `downloadString`) itself strips path components. Task 2 acceptance criterion verifies filename matches `/^\w+-fdpg\.json$/` regex. |
| T-22-13 | DoS / Resource Exhaustion | File picker Import could surface a huge file | MEDIUM | mitigate | Two-layer cap: (1) UI-level check `file.size > MAX_FDPG_FILE_BYTES` BEFORE calling `file.text()` (prevents reading the file into memory); (2) Codec-level check on the JSON text byte-length (Plan 22-02 Task 1) as defence-in-depth. Task 2 acceptance criterion verifies the UI check via the `'enforces 1MB import cap'` test. |
| T-22-14 | Information Disclosure / Wrong Cohort | Row `<Menu>` captures `cohort` by closure; if the cohort list re-renders during menu open (e.g. an import completes while menu is open), the captured `cohort` may be stale | LOW | accept | Mantine `<Menu>` closes on outside-click; rapid concurrent imports are not a realistic race.
