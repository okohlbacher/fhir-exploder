    ```ts
    import { useLocalStorage } from '@mantine/hooks';
    import { useCallback, useEffect, useMemo, useState } from 'react';
    import { notifications } from '@mantine/notifications';
    import {
      COHORTS_STORAGE_KEY, DEFAULT_COHORTS_STORAGE, findActiveCohort,
      type CohortCriterion, type CohortDefinition, type CohortsStorage,
    } from '../quality/cohorts';

    export function useCohorts(): UseCohortsApi {
      const [stored, setStored] = useLocalStorage<CohortsStorage>({
        key: COHORTS_STORAGE_KEY, defaultValue: DEFAULT_COHORTS_STORAGE,
      });
      const [hydrated, setHydrated] = useState(false);
      useEffect(() => { setHydrated(true); }, []);

      const addCohort = useCallback((input: { name: string; criteria: CohortCriterion[] }) => {
        const now = new Date().toISOString();
        const cohort: CohortDefinition = {
          id: crypto.randomUUID(), name: input.name, criteria: input.criteria,
          createdAt: now, updatedAt: now,
        };
        try {
          setStored({ ...stored, cohorts: [...stored.cohorts, cohort] });
        } catch (err) {
          if (err instanceof DOMException && err.name === 'QuotaExceededError') {
            notifications.show({ color: 'red', title: 'Save failed', message: '…' });
          }
          throw err;
        }
        return cohort;
      }, [stored, setStored]);

      const activateCohort = useCallback((id: string | null) => {
        setStored({ ...stored, activeCohortId: id });
      }, [stored, setStored]);

      const activeCohort = useMemo(() => findActiveCohort(stored), [stored]);
      return {
        cohorts: hydrated ? stored.cohorts : [],
        activeCohortId: hydrated ? stored.activeCohortId : null,
        activeCohort: hydrated ? activeCohort : null,
        hydrated, addCohort, activateCohort,
      };
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| React component tree → `useLocalStorage` | Hook reads/writes browser storage; stored values are PHI-adjacent (patient IDs in criteria) |
| cohortResolver → MedplumClient → Blaze | Outbound FHIR queries carry criterion values; inbound bundles carry patient references |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-21-04 | Tampering (localStorage) | `useCohorts` reading corrupted JSON | mitigate | `useLocalStorage` handles parse errors; hook falls back to `DEFAULT_COHORTS_STORAGE` + `console.warn`. Test `legacy migration` verifies shape validation (Plan 21-04). |
| T-21-05 | Injection (URL) | `cohortResolver` criterion → search param | mitigate | All criterion values passed as object keys to `client.searchResourcePages` — Medplum serializes (URLSearchParams under the hood). Never string concatenation into the URL. `system` + `code` joined with `|` only — no user-controlled separator injection possible. |
| T-21-03 | Information disclosure (PHI in logs) | Resolver error path | mitigate | `console.warn` messages reference COUNTS only (`"skipped 3 malformed refs"`), never patient IDs. Notification toast messages never embed the resolved ID list. Verified in acceptance criteria. |
| T-21-02 | DoS (unbounded cohort) | `resolveCriterion` Encounter pagination | mitigate | Resolver loop breaks at `ids.size >= 10_000`. `parsePatientRefs` cap (Plan 21-01) enforces on reference-list input. |
| T-21-06 | DoS (localStorage quota) | `useCohorts.addCohort` write | mitigate | Wrapped in try/catch; QuotaExceededError → red notification toast; rethrow so caller can surface error state. Pattern mirrors `useTrendsHistory.ts:78-93`. |
</threat_model>
