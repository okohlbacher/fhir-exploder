       ```typescript
       import { useEffect, useState } from 'react';
       import type { Resource } from '@medplum/fhirtypes';
       import { useTerminology } from './useTerminology';

       /**
        * Returns a copy of `resource` with all Coding.display fields populated
        * from the terminology server (or cache). The original resource is returned
        * synchronously on first render; the component re-renders once lookups settle.
        * Failures leave display unset; Medplum's formatCoding then falls back to code.
        * Progressive enhancement per D-12: no spinner, no layout shift.
        */
       export function useResolvedResource<T extends Resource>(resource: T | undefined): T | undefined {
         const resolver = useTerminology();
         const [resolved, setResolved] = useState<T | undefined>(resource);

         useEffect(() => {
           if (!resource) {
             setResolved(undefined);
             return;
           }
           setResolved(resource); // render raw immediately
           let cancelled = false;
           resolver
             .resolveResource(resource)
             .then((next) => {
               if (!cancelled) setResolved(next as T);
             })
             .catch(() => {
               // silent per D-07; keep raw resource rendered
             });
           return () => {
             cancelled = true;
           };
         }, [resource, resolver]);

         return resolved;
       }
       ```
