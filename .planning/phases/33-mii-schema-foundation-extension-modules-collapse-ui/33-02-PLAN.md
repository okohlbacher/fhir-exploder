```typescript
// NEW — add to src/utils/mii-modules.ts

/** Returns the list of FHIR resource types this module covers.
 *  Normalises the narrow-schema single string to a one-element array
 *  so downstream code can always iterate. Once plan 33-03 widens
 *  fhirResourceType to `string | string[]`, the array branch activates. */
export function fhirResourceTypesOf(mod: MiiModule): string[] {
  const t = mod.fhirResourceType;
  return Array.isArray(t) ? t : [t];
}

/** Finds the module that owns a given FHIR resource type.
 *  Replaces the inline MII_MODULES.find(m => m.fhirResourceType === type)
 *  pattern at ClinicalTimeline.tsx:85-86 and matches multi-type modules
 *  via Array.includes() once plan 33-03 lands. */
export function findModuleForType(
  type: string,
  modules: MiiModule[] = MII_MODULES,
): MiiModule | undefined {
  return modules.find((m) => fhirResourceTypesOf(m).includes(type));
}

/** Returns the patient search param for a specific (module, type) pair.
 *  Falls back to module-wide `patientSearchParam` when no override exists.
 *  Per D-04, the plan-33-03 widen adds `patientSearchParamOverrides?`;
 *  this helper is ready for that now (checks for the optional map) but
 *  works identically against the narrow schema. */
export function getPatientSearchParamForType(
  mod: MiiModule,
  type: string,
): string {
  // Once plan 33-03 lands, MiiModule has patientSearchParamOverrides?: Record<string, string>
  const overrides = (mod as { patientSearchParamOverrides?: Record<string, string> }).patientSearchParamOverrides;
  return overrides?.[type] ?? mod.patientSearchParam;
}

/** Returns the extra query filter for a specific (module, type) pair.
 *  Per D-05: returns `string | undefined` (NOT `string` with empty fallback).
 *  Callers conditionally append: `if (q) url += '&' + q;`.
 *  Checks `extraQueryByType[type]` first (plan 33-03 widen), falls back
 *  to module-wide `extraQuery`. */
export function getExtraQueryForType(
  mod: MiiModule,
  type: string,
): string | undefined {
  const byType = (mod as { extraQueryByType?: Record<string, string> }).extraQueryByType;
  return byType?.[type] ?? mod.extraQuery;
}
```
