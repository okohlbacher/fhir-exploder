Where `collectPropertyExtensions` is a new module-private helper:

```typescript
interface CollectedExtension {
  url: string;
  ext: Record<string, unknown>;
  /** Source array index for indexed primitives, undefined for scalar primitives or inline. */
  index?: number;
}

function collectPropertyExtensions(value: unknown, sibling: unknown): CollectedExtension[] {
  const out: CollectedExtension[] = [];
  // 1. Scalar primitive — sibling is { extension: Extension[] }
  if (sibling && typeof sibling === 'object' && !Array.isArray(sibling)) {
    const exts = (sibling as { extension?: Array<Record<string, unknown>> }).extension;
    if (Array.isArray(exts)) {
      for (const e of exts) {
        if (e && typeof e === 'object' && typeof e.url === 'string') {
          out.push({ url: e.url, ext: e });
        }
      }
    }
  }
  // 2. Indexed primitive — sibling is array, each slot may be null or { extension: Extension[] }
  if (Array.isArray(sibling)) {
    sibling.forEach((slot, i) => {
      if (slot && typeof slot === 'object') {
        const exts = (slot as { extension?: Array<Record<string, unknown>> }).extension;
        if (Array.isArray(exts)) {
          for (const e of exts) {
            if (e && typeof e === 'object' && typeof e.url === 'string') {
              out.push({ url: e.url, ext: e, index: i });
            }
          }
        }
      }
    });
  }
  // 3. Inline extension on non-primitive (CodeableConcept.extension etc.)
  if (value && typeof value === 'object' && !Array.isArray(value)) {
    const inline = (value as { extension?: Array<Record<string, unknown>> }).extension;
    if (Array.isArray(inline)) {
      for (const e of inline) {
        if (e && typeof e === 'object' && typeof e.url === 'string') {
          out.push({ url: e.url, ext: e });
        }
      }
    }
  }
  return out;
}

export type { CollectedExtension };
```

```typescript
/**
 * ExtensionChip — inline [+N extensions] affordance for property-level FHIR extensions.
 *
 * Phase 47 / READ-02 (D-06, D-07). Closed by default; click expands a list of
 * extensions inline below the property row with each extension's URL + value.
 *
 * Per UI-SPEC §Component 2:
 *   - Closed: <Button variant="subtle" size="xs">+N extension(s)</Button>
 *   - Open:   list of <Code>{url}</Code> + <RenderValue value={ext.value*}/>
 *   - Nested: extension.extension[] recurses with pl='md' indent (D-07)
 */
import { useState } from 'react';
import { Button, Code, Group, Stack, Text } from '@mantine/core';
import { RenderValue, type CollectedExtension } from './ResourcePropertyTable';

export interface ExtensionChipProps {
  extensions: CollectedExtension[];
}

export function ExtensionChip({ extensions }: ExtensionChipProps): JSX.Element | null {
  const [open, setOpen] = useState(false);
  const count = extensions.length;
  if (count === 0) return null;

  const label = count === 1 ? '+1 extension' : `+${count} extensions`;
  const panelId = `ext-chip-panel-${Math.random().toString(36).slice(2, 8)}`;

  return (
    <Stack gap="xs" mt={0}>
      <Button
        variant="subtle"
        size="xs"
        radius="sm"
        onClick={() => setOpen((v) => !v)}
        aria-expanded={open}
        aria-controls={panelId}
      >
        {open ? '−' : '+'} {label.replace(/^\+/, '')}
      </Button>
      {open && (
        <Stack
          id={panelId}
          gap="xs"
          mt="xs"
          pl="md"
          style={{ borderLeft: '2px solid var(--mantine-color-gray-3)' }}
        >
          {extensions.map((c, i) => (
            <ExtensionRow key={i} collected={c} depth={0} />
          ))}
        </Stack>
      )}
    </Stack>
  );
}

interface ExtensionRowProps {
  collected: CollectedExtension;
  depth: number;
}

function ExtensionRow({ collected, depth }: ExtensionRowProps): JSX.Element {
  const { url, ext, index } = collected;
  const value = pickFirstValueX(ext);
  const nested = (ext.extension as Array<Record<string, unknown>> | undefined) ?? [];
  const indent = depth > 0 ? 'md' : 0;

  return (
    <Stack gap={4} pl={indent}>
      <Group gap="xs" wrap="nowrap">
        {typeof index === 'number' && (
          <Text size="xs" c="dimmed">[{index}]</Text>
        )}
        <Code>{url}</Code>
      </Group>
      {value !== undefined && (
        <RenderValue value={value} />
      )}
      {nested.length > 0 && nested.map((n, i) => (
        typeof n === 'object' && n !== null && typeof (n as { url?: string }).url === 'string'
          ? <ExtensionRow key={i} collected={{ url: (n as { url: string }).url, ext: n }} depth={depth + 1} />
          : null
      ))}
    </Stack>
  );
}

/** Extract the first `value*` field from an Extension. Returns undefined if none. */
function pickFirstValueX(ext: Record<string, unknown>): unknown {
  for (const [k, v] of Object.entries(ext)) {
    if (k === 'url') continue;
    if (k === 'extension') continue;
    if (k.startsWith('value')) return v;
  }
  return undefined;
}
```

```typescript
/**
 * ContainedResourcesAccordion — inline render of Resource.contained[].
 *
 * Phase 47 / READ-03 (D-08). Renders below the property table in
 * HumanReadableView, above the bottom resource-level ExtensionsSection.
 *
 * Per UI-SPEC §Component 3:
 *   - Mantine <Accordion multiple variant="separated">; collapsed by default
 *     (no defaultValue per Pitfall 6 — uncontrolled all-collapsed).
 *   - Header: summarizeResource(c).primary (D-08).
 *   - Panel: full <ResourcePropertyTable resource={c} /> (no further drilldown).
 *   - Section heading "Contained Resources" appears only when at least one
 *     contained entry exists (UI-SPEC §When NOT to render).
 *
 * Q2 (RESEARCH.md): terminology walker recursion into contained[] is the
 * existing behavior of useResolvedResource on the parent — verified at
 * task start via grep. Phase 47 does NOT re-wrap each panel in
 * useResolvedResource (A2). HUMAN-UAT confirms no missing displays.
 */
import { Accordion, Stack, Text } from '@mantine/core';
import type { Resource } from '@medplum/fhirtypes';
import { summarizeResource } from '../../utils/summarizeResource';
import { ResourcePropertyTable } from './ResourcePropertyTable';

export interface ContainedResourcesAccordionProps {
  resource: Resource;
}

export function ContainedResourcesAccordion({ resource }: ContainedResourcesAccordionProps): JSX.Element | null {
  const contained = (resource as { contained?: Resource[] }).contained;
  if (!Array.isArray(contained) || contained.length === 0) return null;

  return (
    <Stack gap="xs">
      <Text size="sm" c="dimmed" tt="uppercase" mt="lg" mb="xs">
        Contained Resources
      </Text>
      <Accordion multiple variant="separated" radius="sm">
        {contained.map((c, i) => {
          const value = c.id ?? `idx-${i}`;
          const summary = summarizeResource(c).primary;
          return (
            <Accordion.Item key={value} value={value}>
              <Accordion.Control>
                <Text size="sm" fw={500}>
                  {c.resourceType} — {summary || '(no summary)'}
                </Text>
              </Accordion.Control>
              <Accordion.Panel>
                <ResourcePropertyTable resource={c} parentResource={resource} />
              </Accordion.Panel>
            </Accordion.Item>
          );
        })}
      </Accordion>
    </Stack>
  );
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| FHIR Resource → ResourcePropertyTable property walker | `_propertyName.extension[]` arrays and `value*` fields are server-supplied and may contain hostile content (HTML strings, script payloads, oversized arrays). |
| FHIR Resource.contained[] → ContainedResourcesAccordion | Contained resources are server-supplied; arbitrarily deeply nested structures are possible (Pitfall: pathological contained arrays). |
| Extension value rendering → DOM | Extension `value*` fields rendered via `<RenderValue>` must use Mantine text-node escaping, never `dangerouslySetInnerHTML`. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-47-02 | DoS | ContainedResourcesAccordion (pathological contained[] count) | accept | Mantine `<Accordion>` is virtualization-free but renders only headers initially (collapsed-by-default per D-08 + Pitfall 6). Panels are mounted lazily by Mantine when expanded. For N=100 contained resources, the initial render is bounded by N header strings (cheap). For N=10000, a defensive `contained.slice(0, 200)` cap could be added but Phase 47 does not include it — flagged in HUMAN-UAT. Acceptable for current single-server local-tool threat model; revisit at Phase 49 graph if observed. |
| T-47-03 | Tampering / XSS | ExtensionChip / ExtensionRow value rendering | mitigate | All extension `value*` rendering routes through the local `<RenderValue>` (exported from `ResourcePropertyTable.tsx`), which uses Mantine `<Text>{string}</Text>` and `<Code>{string}</Code>` — both rely on React's text-node escaping. Verified via grep: `! grep -q "dangerouslySetInnerHTML" src/components/explorer/ExtensionChip.tsx src/components/explorer/ContainedResourcesAccordion.tsx`. URLs in extension `url` field are rendered as text inside `<Code>` — never as anchor `href`. |
| T-47-07 | Tampering | ResourcePropertyTable property walker (sibling lookup `_K`) | mitigate | The walker uses bracket access `(resource as Record<string, unknown>)[\`_${key}\`]` — same defensive pattern as `summarizeResource.summarizeGeneric` (Phase 46 T-46-04).
