```tsx
  // QUAL-02 (D-12/D-13): 3-stop gradient via Mantine theme tokens.
  // green ≥100%, yellow at threshold..<100%, red below threshold.
  // When threshold is null (metric disabled), fall back to green-when-100,
  // no-color otherwise.
  const threshold = getActiveThreshold(metricKey);
  let bg: string | undefined;
  let fg: string | undefined;
  if (value >= 100) {
    bg = 'var(--mantine-color-green-1)';
    fg = 'var(--mantine-color-green-9)';
  } else if (threshold !== null && value >= threshold) {
    bg = 'var(--mantine-color-yellow-1)';

  } else if (threshold !== null && value < threshold) {
    bg = 'var(--mantine-color-red-1)';
    fg = 'var(--mantine-color-red-9)';
  }

  // For Progress bar: green at ≥100, red when below threshold, yellow between.
  const barColor =
    value >= 100 ? 'green'
    : threshold !== null && value < threshold ? 'red'
    : threshold !== null && value >= threshold ? 'yellow'
    : 'indigo';

  return (
    <Table.Td style={bg ? { backgroundColor: bg } : undefined}>
      <Stack gap={2}>
        <Text
          size="sm"
          fw={600}
          c={fg}

          {value}%
        </Text>
```

```ts
/** QUAL-03: CSV-escape one cell — wrap in quotes if it contains comma, quote, or newline. */
function csvEscape(value: string): string {
  if (value.includes(',') || value.includes('"') || value.includes('\n')) {
    return `"${value.replace(/"/g, '""')}"`;
  }
  return value;
}

/** QUAL-03: Build CSV body from matrix rows (post-sort, post-filter).
 *  Sparse cells render as empty string. Skips loading rows.
 *  Result is prefixed with UTF-8 BOM and ready for download.
 *  Exported for unit tests.
 */
export function buildMatrixCsv(rows: ReadonlyArray<MatrixRow>): string {
  const header = ['Resource type', 'Complete %', 'Coverage %', 'Validation %', 'References %', 'Dup', 'Issues'];
  const lines: string[] = [header.join(',')];
  for (const row of rows) {
    if (row.countLoading) continue; // skip skeleton placeholders
    const cells = [
      csvEscape(row.type),
      row.completeness !== undefined ? String(row.completeness) : '',
      row.coverage !== undefined ? String(row.coverage) : '',
      row.validation !== undefined ? String(row.validation) : '',
      row.references !== undefined ? String(row.references) : '',
      row.dup !== undefined ? String(row.dup) : '',
      row.issues !== undefined ? String(row.issues) : '',
    ];
    lines.push(cells.join(','));
  }
  return '﻿' + lines.join('\n');
}

/** QUAL-03: Filename helper — sanitize host + UTC date.
 *  Exported for unit tests.
 */
export function buildMatrixCsvFilename(serverUrl: string, now: Date = new Date()): string {
  let host = 'unknown';
  try {
    host = new URL(serverUrl).host;
  } catch {
    host = 'unknown';
  }
  const safeHost = host.toLowerCase().replace(/[^a-z0-9.-]/g, '-');
  const date = now.toISOString().slice(0, 10);
  return `quality-matrix-${safeHost}-${date}.csv`;
}
```
